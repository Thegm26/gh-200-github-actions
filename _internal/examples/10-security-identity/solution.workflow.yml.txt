name: safe review and non-deploying identity exercise
on: pull_request
permissions:
  contents: read
jobs:
  inspect:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@d23441a48e516b6c34aea4fa41551a30e30af803 # reviewed full SHA
      - name: Treat title as data
        env:
          PR_TITLE: ${{ github.event.pull_request.title }}
        run: printf '%s\n' "$PR_TITLE"
  identity-notes:
    if: ${{ github.event.pull_request.head.repo.fork == false }}
    needs: inspect
    runs-on: ubuntu-latest
    environment: production
    permissions:
      contents: read
      id-token: write
    steps:
      - name: Explain only; do not request or exchange a token
        run: echo 'A reviewed provider login action would exchange OIDC here.' >> "$GITHUB_STEP_SUMMARY"
