name: GH-200 lab 04-context-as-data
on:
  workflow_dispatch:
    inputs:
      message:
        description: Maintainer-controlled teaching sample
        required: true
        default: '$(printf changed)'
permissions:
  contents: read
jobs:
  lab:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - name: Unsafe direct interpolation (teaching starter)
        shell: bash
        run: |
          actual="${{ inputs.message }}"
          printf '%s\n' "$actual"
          test "$actual" = '$(printf changed)'
