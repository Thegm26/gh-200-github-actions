name: GH-200 lab 11-failure-evidence
on:
  workflow_dispatch: {}
permissions:
  contents: read
jobs:
  lab:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - name: Test
        shell: bash
        run: |
          echo intentional-test-failure
          exit 1
      - name: Evidence (skipped after failure)
        shell: bash
        run: echo test-finished >> "$GITHUB_STEP_SUMMARY"
