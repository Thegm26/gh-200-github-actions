name: GH-200 lab 16
on:
  workflow_dispatch: {}
permissions:
  contents: read
jobs:
  artifact:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - shell: bash
        run: mkdir -p dist && printf artifact > dist/result.txt
      - uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: gh200-lab-output
          path: dist/result.txt
          if-no-files-found: error
      - uses: actions/download-artifact@d3f86a106a0bac45b974a628896c90dbdf5c8093
        with:
          name: gh200-lab-output
          path: recovered
      - shell: bash
        run: test "$(cat recovered/result.txt)" = artifact
