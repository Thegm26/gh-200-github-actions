  workflow_dispatch: {}

  contents: read

  node:
    runs-on: ubuntu-latest
    timeout-minutes: 5

        with:
          node-version: 20

        env:
          REQUESTED_NODE: ${{ matrix.node }}
        run: test "$(node -p "process.versions.node.split('.')[0]")" = "$REQUESTED_NODE"
