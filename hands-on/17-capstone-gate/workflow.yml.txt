name: GH-200 lab 17
on:
  workflow_dispatch: {}
permissions:
  contents: read
jobs:
  build:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    outputs:
      result: ${{ steps.result.outputs.result }}
    steps:
      - id: result
        shell: bash
        run: echo "result=not-ready" >> "$GITHUB_OUTPUT"
  deploy:
    needs: build
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - shell: bash
        run: test "${{ needs.build.outputs.result }}" = ready
