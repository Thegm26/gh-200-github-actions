name: safe inactive final capstone solution
on:
  pull_request:
    branches: [main]
    paths: ["src/**", "test/**"]
  workflow_dispatch:
    inputs:
      run_preview:
        description: Run safe training preview
        required: true
        default: false
        type: boolean
permissions:
  contents: read
jobs:
  prepare:
    runs-on: ubuntu-latest
    outputs:
      version: ${{ steps.meta.outputs.version }}
    steps:
      - id: meta
        run: echo "version=9.0" >> "$GITHUB_OUTPUT"
  test:
    needs: prepare
    runs-on: ${{ matrix.os }}
    strategy:
      fail-fast: false
      max-parallel: 2
      matrix:
        os: [ubuntu-latest, windows-latest]
        node: [20, 22]
        exclude: [{ os: windows-latest, node: 22 }]
    steps:
      - uses: actions/checkout@d23441a48e516b6c34aea4fa41551a30e30af803
      - name: Treat event text as data
        env:
          PR_TITLE: ${{ github.event.pull_request.title || 'manual-preview' }}
        shell: bash
        run: printf '%s\n' "$PR_TITLE"
      - shell: bash
        run: echo "${{ needs.prepare.outputs.version }}" > report.txt
      - uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: report-${{ matrix.os }}-${{ matrix.node }}
          path: report.txt
          retention-days: 7
  risk:
    needs: prepare
    uses: ./.github/workflows/reusable-risk.yml
    with:
      version: ${{ needs.prepare.outputs.version }}
  deployment-review-only:
    if: ${{ github.event_name == 'workflow_dispatch' && inputs.run_preview }}
    needs: [test, risk]
    runs-on: [self-hosted, linux, finance-production]
    environment: production
    permissions:
      contents: read
      id-token: write
    steps:
      - run: >-
          echo 'No deployment: reusable risk is ${{ needs.risk.outputs.risk }}; verify attestation identity and digest before any authorized promotion.'
