name: data boundary solution
on: workflow_dispatch
jobs:
  prepare:
    runs-on: ubuntu-latest
    outputs:
      version: ${{ steps.meta.outputs.version }}
    steps:
      - id: meta
        run: echo "version=2.4.0" >> "$GITHUB_OUTPUT"
  test-linux:
    needs: prepare
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      max-parallel: 2
      matrix:
        node: [20, 22]
        include:
          - { node: 24, experimental: true }
    services:
      postgres:
        image: postgres:16
        env:
          POSTGRES_PASSWORD: training
        ports: [5432:5432]
        options: >-
          --health-cmd "pg_isready -U postgres"
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
    steps:
      - run: echo "MODE=integration" >> "$GITHUB_ENV"
      - run: printf '%s %s\n' "${{ needs.prepare.outputs.version }}" "$MODE"
      - run: echo "## ubuntu-latest / ${{ matrix.node }}" >> "$GITHUB_STEP_SUMMARY"
  test-windows:
    needs: prepare
    runs-on: windows-latest
    strategy:
      fail-fast: false
      max-parallel: 2
      matrix:
        node: [20]
    steps:
      - shell: pwsh
        run: '"MODE=integration" | Out-File -FilePath $env:GITHUB_ENV -Append'
      - shell: pwsh
        run: '"${{ needs.prepare.outputs.version }} $env:MODE"'
      - shell: pwsh
        run: '"## windows-latest / ${{ matrix.node }}" | Out-File -FilePath $env:GITHUB_STEP_SUMMARY -Append'
