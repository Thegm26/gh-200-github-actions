name: repair triggers
on:
  push:
    branches: [main]
    paths: ["src/**"]
  workflow_dispatch:
    inputs:
      deploy:
        description: 'deploy'
        required: true
        type: boolean
      target:
        description: 'The target'
        required: true
        default: 'target1'
        type: choice
        options:
          - target1
          - target2
          - target3
defaults: &run-defaults
  run:
    shell: bash
    working-directory: .
jobs:
  inspect:
    if: ${{github.event_name == 'push' || inputs.deploy == true }}
    runs-on: ubuntu-latest
    defaults: *run-defaults
    steps:
      - name: unsafe placeholder
        env:
          EVENT_TEXT: ${{github.event.pull_request.title || 'manual-or-push'}}
        run: printf '%s\n' "$EVENT_TEXT"
